# @kotobaya/shell-client@1.0.0

A TypeScript SDK client for the localhost API.

## Usage

First, install the SDK from npm.

```bash
npm install @kotobaya/shell-client --save
```

Next, try it out.


```ts
import {
  Configuration,
  DeviceApi,
} from '@kotobaya/shell-client';
import type { GetAnkiConnectStateRequest } from '@kotobaya/shell-client';

async function example() {
  console.log("🚀 Testing @kotobaya/shell-client SDK...");
  const api = new DeviceApi();

  try {
    const data = await api.getAnkiConnectState();
    console.log(data);
  } catch (error) {
    console.error(error);
  }
}

// Run the test
example().catch(console.error);
```
